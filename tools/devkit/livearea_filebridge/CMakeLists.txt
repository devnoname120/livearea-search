cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(NOT DEFINED ENV{VITASDK})
    set(ENV{VITASDK} /usr/local/vitasdk)
  endif()
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain")
endif()
project(livearea_search_filebridge C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)

add_executable(filebridge bridge.c)
target_compile_options(filebridge PRIVATE -O2 -Wall -Wextra -Werror -fno-tree-vectorize)
target_link_libraries(filebridge SceIofilemgr_stub SceLibKernel_stub)
set_target_properties(filebridge PROPERTIES OUTPUT_NAME filebridge.elf)
set(VITA_MAKE_FSELF_FLAGS "-a 0x2808000000000000")
vita_create_self(filebridge.bin filebridge UNSAFE NOASLR)
